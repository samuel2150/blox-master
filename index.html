async function consultarFrutas() { const base = $("fruitApi").value.trim().replace(/\/+$/, ""); const fruta = $("fruitName").value.trim().toLowerCase(); if (!base) { $("fruitHint").textContent = "Configure primeiro a URL de uma API de frutas compatível."; registrar("Frutas: nenhuma API configurada."); return; } if (!base.startsWith("https://")) { $("fruitHint").textContent = "Por segurança, use uma URL HTTPS."; registrar("Frutas: a URL precisa usar HTTPS."); return; } $("fruitButton").disabled = true; $("fruitHint").textContent = "Consultando a API..."; try { const tarefas = [ consultarJSON(base + "/info/stock") ]; if (fruta) { tarefas.push( consultarJSON( base + "/info/fruits/" + encodeURIComponent(fruta) ) ); } const resultados = await Promise.allSettled(tarefas); if (resultados[0].status === "fulfilled") { $("fruitStock").textContent = JSON.stringify( resultados[0].value.dados ); registrar("Stock recebido da API configurada."); } else { $("fruitStock").textContent = "Falha na consulta"; registrar( "Erro no stock: " + resultados[0].reason.message ); } if (fruta && resultados[1].status === "fulfilled") { $("fruitInfo").textContent = JSON.stringify( resultados[1].value.dados ); registrar("Dados da fruta recebidos da API."); } else if (fruta) { $("fruitInfo").textContent = "Falha na consulta"; registrar( "Erro na fruta: " + (resultados[1]?.reason?.message || "Resposta inválida") ); } else { $("fruitInfo").textContent = "Digite uma fruta"; } $("fruitHint").textContent = "Consulta concluída. Verifique a origem e a atualização dos dados."; } catch (erro) { $("fruitHint").textContent = "Não foi possível consultar a API: " + erro.message; registrar("Erro na integração de frutas: " + erro.message); } finally { $("fruitButton").disabled = false; } } $("fruitButton").addEventListener("click", consultarFrutas);
